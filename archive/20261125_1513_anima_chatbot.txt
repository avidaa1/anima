function jaccard(a,b){
  var setA=new Set(a), setB=new Set(b);
  var inter=0; setA.forEach(function(w){ if(setB.has(w)) inter++; });
  var union=new Set(a.concat(b)).size;
  return union===0?0:inter/union;
}
